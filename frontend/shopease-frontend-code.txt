import {
  createContext,
  useContext,
  useEffect,
  useState,
} from "react";

import {
  registerUser,
  loginUser,
  getMe,
  logoutUser,
} from "../services/authApi";

const AuthContext = createContext(null);

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);

  const [token, setToken] = useState(
    localStorage.getItem("shopease-token")
  );

  const [loading, setLoading] = useState(true);

  // Restore logged-in user after page refresh
  useEffect(() => {
    const loadUser = async () => {
      if (!token) {
        setLoading(false);
        return;
      }

      try {
        const data = await getMe(token);
        setUser(data.user);
      } catch (error) {
        localStorage.removeItem("shopease-token");
        setToken(null);
        setUser(null);
      } finally {
        setLoading(false);
      }
    };

    loadUser();
  }, [token]);

  const register = async (userData) => {
    return await registerUser(userData);
  };

  const login = async (userData) => {
    const data = await loginUser(userData);

    localStorage.setItem("shopease-token", data.token);

    setToken(data.token);
    setUser(data.user);

    return data;
  };

  const logout = async () => {
    try {
      if (token) {
        await logoutUser(token);
      }
    } finally {
      localStorage.removeItem("shopease-token");
      setToken(null);
      setUser(null);
    }
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        token,
        loading,
        register,
        login,
        logout,
        isAuthenticated: !!user,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  return useContext(AuthContext);
};

import {
  createContext,
  useContext,
  useEffect,
  useState,
} from "react";

import { useAuth } from "./AuthContext";
import {
  getCart,
  addToCart,
  updateCartItem,
  removeCartItem,
} from "../services/cartApi";

const CartContext = createContext(null);

export const CartProvider = ({ children }) => {
  const [cart, setCart] = useState(null);

  const { token } = useAuth();

  // Load the logged-in user's cart
  useEffect(() => {
    const loadCart = async () => {
      // If user is not logged in, clear cart state
      if (!token) {
        setCart(null);
        return;
      }

      try {
        const data = await getCart(token);
        setCart(data.cart);
      } catch (error) {
        console.error("Load cart error:", error);
      }
    };

    loadCart();
  }, [token]);

  // Add a product to the cart
  const addItem = async (productId, quantity = 1) => {
    if (!token) {
      throw new Error("Please login to add products to cart");
    }

    const data = await addToCart(
      productId,
      quantity,
      token
    );

    setCart(data.cart);

    return data;
  };

  // Update product quantity in the cart
  const updateItem = async (productId, quantity) => {
    if (!token) {
      throw new Error("Please login to update cart");
    }

    const data = await updateCartItem(
      productId,
      quantity,
      token
    );

    setCart(data.cart);

    return data;
  };

  // Remove a product from the cart
  const removeItem = async (productId) => {
    if (!token) {
      throw new Error(
        "Please login to remove products from cart"
      );
    }

    const data = await removeCartItem(
      productId,
      token
    );

    setCart(data.cart);

    return data;
  };

  return (
    <CartContext.Provider
      value={{
        cart,
        setCart,
        token,
        addItem,
        updateItem,
        removeItem,
      }}
    >
      {children}
    </CartContext.Provider>
  );
};

export const useCart = () => {
  return useContext(CartContext);
};

import { useState } from "react";
import { useNavigate } from "react-router-dom";
import { useAuth } from "../context/AuthContext";

function Login() {
  const { login } = useAuth();
  const navigate = useNavigate();

  const [formData, setFormData] = useState({
    email: "",
    password: "",
  });

  const [message, setMessage] = useState("");

  const handleChange = (e) => {
    setFormData({
      ...formData,
      [e.target.name]: e.target.value,
    });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    try {
      const data = await login(formData);

      setMessage(data.message);

      // Redirect according to user role
      if (data.user.role === "admin") {
        navigate("/admin");
      } else {
        navigate("/");
      }
    } catch (error) {
      setMessage(
        error.response?.data?.message || "Login failed"
      );
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center">
      <form
        onSubmit={handleSubmit}
        className="w-full max-w-md space-y-4 border p-6 rounded-lg"
      >
        <h1 className="text-3xl font-bold">Login</h1>

        <input
          type="email"
          name="email"
          placeholder="Email"
          value={formData.email}
          onChange={handleChange}

        <input

import { useNavigate } from "react-router-dom";
import { useCart } from "../context/CartContext";

function Cart() {
  const { cart, updateItem, removeItem } = useCart();
  const navigate = useNavigate();

  const handleIncrease = async (item) => {
    try {
      await updateItem(
        item.product._id,
        item.quantity + 1
      );
    } catch (error) {
      alert(
        error.response?.data?.message ||
          error.message ||
          "Failed to update quantity"
      );
    }
  };

  const handleDecrease = async (item) => {
    if (item.quantity <= 1) {
      return;
    }

    try {
      await updateItem(
        item.product._id,
        item.quantity - 1
      );
    } catch (error) {
      alert(
        error.response?.data?.message ||
          error.message ||
          "Failed to update quantity"
      );
    }
  };

  const handleRemove = async (item) => {
    try {
      await removeItem(item.product._id);
    } catch (error) {
      alert(
        error.response?.data?.message ||
          error.message ||
          "Failed to remove product"
      );
    }
  };

  const handleCheckout = () => {
    navigate("/checkout");
  };

  if (!cart) {
    return <p className="p-6">Loading cart...</p>;
  }

  if (cart.items.length === 0) {
    return (
      <div className="p-6">
        <h1 className="text-3xl font-bold mb-4">
          My Cart
        </h1>

        <p>Your cart is empty.</p>
      </div>
    );
  }

  // Calculate total price of all cart items
  const cartTotal = cart.items.reduce(
    (total, item) =>
      total + item.product.price * item.quantity,
    0
  );

  return (
    <div className="p-6">
      <h1 className="text-3xl font-bold mb-6">
        My Cart
      </h1>

      <div className="space-y-4">
        {cart.items.map((item) => (
          <div
            key={item.product._id}
            className="border rounded-lg p-4"
          >
            <h2 className="text-xl font-semibold">
              {item.product.name}
            </h2>

            <p className="mt-2">
              Price: Rs. {item.product.price}
            </p>

            <p>
              Quantity: {item.quantity}
            </p>

            <div className="flex gap-2 mt-2">
              <button
                type="button"
                onClick={() => handleDecrease(item)}
                disabled={item.quantity <= 1}
                className="bg-black text-white px-3 py-1 rounded disabled:bg-gray-400 disabled:cursor-not-allowed"
              >
                -
              </button>

              <button
                type="button"
                onClick={() => handleIncrease(item)}
                disabled={
                  item.quantity >= item.product.stock
                }
                className="bg-black text-white px-3 py-1 rounded disabled:bg-gray-400 disabled:cursor-not-allowed"
              >
                +
              </button>

              <button
                type="button"
                onClick={() => handleRemove(item)}
                className="bg-red-600 text-white px-4 py-1 rounded"
              >
                Remove
              </button>
            </div>

            <p className="font-bold mt-2">
              Subtotal: Rs.{" "}
              {item.product.price * item.quantity}
            </p>
          </div>
        ))}
      </div>

      <div className="mt-6 border-t pt-4">
        <p className="text-2xl font-bold">
          Total: Rs. {cartTotal}
        </p>

        <button
          type="button"
          onClick={handleCheckout}
          className="mt-4 bg-black text-white px-6 py-2 rounded"
        >
          Proceed to Checkout
        </button>
      </div>
    </div>
  );
}

export default Cart;

        token
      );

      // Backend clears the database cart.

    price: "",
    category: "",

    image: "",
  });

    setEditingProductId(null);

      name: "",
      description: "",
      price: "",
      category: "",
      stock: "",
      image: "",
    });

    setFormData((previousData) => ({
      ...previousData,
      [e.target.name]: e.target.value,

    const file = e.target.files?.[0];

    if (!file) {
      setSelectedImage(null);
      return;
    }

    setSelectedImage(file);
    setError("");
  };

  const handleEdit = (product) => {
    setEditingProductId(product._id);
    setSelectedImage(null);

    setFormData({
      name: product.name,
      description: product.description,
      price: product.price,
      category: product.category,
      stock: product.stock,
      image: product.image || "",
    });

    setError("");
    setSuccessMessage("");

    window.scrollTo({
      top: 0,
      behavior: "smooth",
    });
  };

  const handleCancelEdit = () => {
    resetForm();
    setError("");
    setSuccessMessage("");
  };

  const handleSubmit = async (e) => {
    e.preventDefault();

    try {
      setSaving(true);
      setError("");
      setSuccessMessage("");

      let imageUrl = formData.image;

      // Upload selected image first
      if (selectedImage) {
        setUploadingImage(true);

        const uploadData =
          await uploadProductImage(
            selectedImage,
            token
          );

        imageUrl = uploadData.imageUrl;

        setUploadingImage(false);
      }

      const productData = {
        ...formData,
        image: imageUrl,
      };

      if (editingProductId) {
        const data = await updateProduct(
          editingProductId,
          productData,
          token
        );

        setProducts((currentProducts) =>
          currentProducts.map((product) =>
            product._id === editingProductId
              ? data.product
              : product
          )
        );

        setSuccessMessage(

        );
      } else {

        ]);

import {
  useEffect,
  useState,
} from "react";

import {
  useNavigate,
  useParams,
} from "react-router-dom";

import { useAuth } from "../context/AuthContext";

import {
  getAdminOrderById,
  updateOrderStatus,
} from "../services/adminOrderApi";

const statusTransitions = {
  Pending: ["Confirmed", "Cancelled"],
  Confirmed: ["Processing", "Cancelled"],
  Processing: ["Shipped", "Cancelled"],
  Shipped: ["Delivered"],
  Delivered: [],
  Cancelled: [],
};

function AdminOrderDetails() {
  const { id } = useParams();
  const { token } = useAuth();
  const navigate = useNavigate();

  const [order, setOrder] = useState(null);
  const [selectedStatus, setSelectedStatus] =
    useState("");
  const [loading, setLoading] = useState(true);
  const [updating, setUpdating] = useState(false);

  const [errorMessage, setErrorMessage] =
    useState("");

  const [successMessage, setSuccessMessage] =
    useState("");

  useEffect(() => {
    const loadOrder = async () => {
      try {
        setLoading(true);
        setErrorMessage("");

        const data = await getAdminOrderById(
          id,
          token
        );

        setOrder(data.order);
        setSelectedStatus("");
      } catch (error) {
        setErrorMessage(
          error.response?.data?.message ||
            error.message ||
            "Failed to load order"
        );
      } finally {
        setLoading(false);
      }
    };

    if (token && id) {
      loadOrder();
    }
  }, [id, token]);

  const handleStatusUpdate = async () => {
    if (!selectedStatus) {
      setErrorMessage(
        "Please select a new order status"
      );
      return;
    }

    try {
      setUpdating(true);
      setErrorMessage("");
      setSuccessMessage("");

      const data = await updateOrderStatus(
        id,
        selectedStatus,
        token
      );

      setOrder(data.order);
      setSelectedStatus("");

      setSuccessMessage(data.message);
    } catch (error) {
      setErrorMessage(
        error.response?.data?.message ||
          error.message ||
          "Failed to update order status"
      );
    } finally {
      setUpdating(false);
    }
  };

  if (loading) {
    return (
      <p className="p-6">
        Loading order...
      </p>
    );
  }

  if (!order) {
    return (
      <div className="p-6">
        <p>{errorMessage || "Order not found"}</p>

        <button
          type="button"
          onClick={() =>
            navigate("/admin/orders")
          }
          className="mt-4 bg-black text-white px-5 py-2 rounded"
        >
          Back to Orders
        </button>
      </div>
    );
  }

  const availableStatuses =
    statusTransitions[order.status] || [];

  return (
    <div className="p-6">
      <div className="max-w-4xl mx-auto">
        <button
          type="button"
          onClick={() =>
            navigate("/admin/orders")
          }
          className="border px-4 py-2 rounded mb-5"
        >
          ← Back to Admin Orders
        </button>

        <h1 className="text-3xl font-bold">
          Manage Order
        </h1>

        <p className="text-sm text-gray-600 break-all mt-2">
          Order ID: {order._id}
        </p>

        {successMessage && (
          <div className="border border-green-500 bg-green-50 text-green-700 p-3 rounded mt-5">
            {successMessage}
          </div>
        )}

        {errorMessage && (
          <div className="border border-red-500 bg-red-50 text-red-700 p-3 rounded mt-5">
            {errorMessage}
          </div>
        )}

        {/* Customer */}
        <div className="border rounded-lg p-4 mt-6">
          <h2 className="text-xl font-bold mb-3">
            Customer
          </h2>

          <p>
            Name: {order.user?.name || "N/A"}
          </p>

          <p>
            Email: {order.user?.email || "N/A"}
          </p>
        </div>

        {/* Order Information */}
        <div className="border rounded-lg p-4 mt-4">
          <h2 className="text-xl font-bold mb-3">
            Order Information
          </h2>

          <p>
            Status:{" "}
            <span className="font-bold">
              {order.status}
            </span>
          </p>

          <p>
            Payment: {order.paymentMethod}
          </p>

          <p>
            Date:{" "}
            {new Date(
              order.createdAt
            ).toLocaleString()}
          </p>
        </div>

        {/* Status Management */}
        <div className="border rounded-lg p-4 mt-4">
          <h2 className="text-xl font-bold mb-3">
            Update Status
          </h2>

          {availableStatuses.length > 0 ? (
            <>
              <select
                value={selectedStatus}
                onChange={(e) =>
                  setSelectedStatus(
                    e.target.value
                  )
                }
                disabled={updating}
                className="w-full border rounded px-3 py-2"
              >
                <option value="">
                  Select new status
                </option>

                {availableStatuses.map(
                  (status) => (
                    <option
                      key={status}
                      value={status}
                    >
                      {status}
                    </option>
                  )
                )}
              </select>

              <button
                type="button"
                onClick={handleStatusUpdate}
                disabled={
                  updating || !selectedStatus
                }
                className="mt-3 bg-black text-white px-5 py-2 rounded disabled:bg-gray-400 disabled:cursor-not-allowed"
              >
                {updating
                  ? "Updating..."
                  : "Update Status"}
              </button>
            </>
          ) : (
            <p className="text-gray-600">
              This order has reached its final
              status and cannot be changed.
            </p>
          )}
        </div>

        {/* Products */}
        <div className="border rounded-lg p-4 mt-4">
          <h2 className="text-xl font-bold mb-4">
            Products
          </h2>

          <div className="space-y-4">
            {order.items.map((item) => (
              <div
                key={item.product}
                className="border-b pb-3"
              >
                <p className="font-semibold">
                  {item.productName}
                </p>

                <p>
                  Price: Rs. {item.price}
                </p>

                <p>
                  Quantity: {item.quantity}
                </p>

                <p className="font-semibold">
                  Subtotal: Rs.{" "}
                  {item.price *
                    item.quantity}
                </p>
              </div>
            ))}
          </div>

          <p className="text-xl font-bold mt-4">
            Total: Rs. {order.totalAmount}
          </p>
        </div>

        {/* Delivery */}
        <div className="border rounded-lg p-4 mt-4">
          <h2 className="text-xl font-bold mb-3">
            Delivery Information
          </h2>

          <p>
            Name:{" "}
            {order.deliveryInfo.fullName}
          </p>

          <p>
            Phone:{" "}
            {order.deliveryInfo.phone}
          </p>

          <p>
            Address:{" "}
            {order.deliveryInfo.address}
          </p>

          <p>
            City: {order.deliveryInfo.city}
          </p>
        </div>
      </div>
    </div>
  );
}

export default AdminOrderDetails;
